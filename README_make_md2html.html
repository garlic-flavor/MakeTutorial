make md2html

試験実装

概要

MarkdownからHTMLを生成する。

動機

  • MacOSでMarkdownを開くデフォルトアプリをXCodeから変えられない。 そしてXCodeは起動が遅い。
  • cmark-gfmの出力はコードのシンタックスハイライトがないので、 そういうものを付ける。
  • とりあえずHTML生成に対応しておけば何かいいことがあるかも知れない。

目標

  • 単一でJavascriptを使用しないHTMLを生成する。

謝辞

  • cmark-gfmを使っています。
sh
BACKEND="cmark-gfm -e table -e autolink"

使い方

sh
USAGE="MakeMD2HTML: MarkdownからHTMLを生成する。
使い方:
>$0 [-o out.html] [-b cmark-gfm] file.md
オプション:
  -o : 出力ファイルを指定する。初期値では標準出力。
  -t : HTMLのタイトルを指定する。
  -b : バックエンドを指定する。初期値は \"$BACKEND\"
  -h : このヘルプを表示する。
"
typeset -g output
#            ┗出力ファイル名

バッファが空なら埋める。

もしバッファが空なら、改行を出力し、バッファに読み込む。

Params:

  • -n: 改行を挿入しない。

Return:

  • 読み込めるものがなければ1を返す。
sh
#------------------------------------------------------------------------------
function feedBuffer {
  local opt nonewline
  while getopts n opt; do
    case $opt in
      n) nonewline=true ;;
      *) :;;
    esac
  done
  (( 1 < OPTIND )) && shift $((OPTIND - 1))
  if [[ -z $buf ]] then
    if [[ ! $nonewline ]]; then
      print -
    fi
    if ! IFS= read -r buf; then
      return 1
    fi
  fi
}

とりあえず一文字処理する。

Params:

  • $1 : 処理する文字数を指定する。
sh
#------------------------------------------------------------------------------
function processChar {
  local n=${1:-1}
  #         ┗第一引数が無ければ1を代入
  print -rn - ${buf:0:$n}
  #         ┗ printのオプションをここまでとする。'-'一文字を出力する際に必要。
  buf=${buf:$n}
  #        ┗ $n 文字目より後のスライスを代入
}

もし先頭が$1なら処理する。

Params:

  • $1 : テストする文字列
  • $2 : もし真だったら先にこれも出力する。
  • $3 : もし真だったら後にこれも出力する。

Return:

  • $1で始まっていなけれは1を返す。
sh
#------------------------------------------------------------------------------
function processIf {
  if [[ $buf != $1* ]]; then return 1; fi
  #               ┗後ろに任意の文字が続いていても良い
  print -rn - $2
  processChar ${#1}
  print -rn - $3
}

要素を開始する。

Params:

  • $1 : テストする文字列

Return:

  • $1で始まっていなければ1を返す。
sh
#------------------------------------------------------------------------------
function processStartSpanIf {
  processIf $1 "<span class=\"$2\">"
  return $?
}

Params:

  • $1 : テストする文字列。
sh
#------------------------------------------------------------------------------
function processEndSpanIf {
  processIf $1 '' '</span>'
  return $?
}

正規表現にマッチしたら処理する。

Params:

  • $1 : テストする正規表現
  • $2 : もしマッチしていたらマッチした文字列の前に出力する。
  • $3 : もしマッチしていたらマッチした文字列の後に出力する。
sh
#------------------------------------------------------------------------------
function processMatch {
  if [[ ! $buf =~ $1 ]]; then return 1; fi
  #             ┗正規表現でテストする。
  print -rn - $2
  processChar ${#MATCH}
  #               ┗マッチの結果が格納されている。
  print -rn - $3
}

function processEntity {

一行コメントを処理する。

Params:

  • $1 : 一行コメントを開始するパターン。初期値は'#'
  • $2 : コメントはspan要素で出力される。その際のclassを指定する。 初期値はcomment

Return:

  • コメント行でなければ1を返す。
sh
#------------------------------------------------------------------------------
function processOneLineComment {
  local header=${1:-'#'} class=${2:-comment}
  if ! processIf $header "<span class=\"$class\">"; then return 1; fi
  print -rn - $buf"</span>"
  buf=
}

最も単純なエスケープシーケンスを処理する。

Return:

  • バックスラッシュでなければ1を返す。
sh
#------------------------------------------------------------------------------
function processSimpleEscapeSequence {
  if ! processIf "\\"; then return 1; fi
  processChar
}

シンプルな文字列を処理する。

Params:

  • $1 : クォート文字。ダブルクォートは文字実体参照により&quot;になっている ので注意。
  • $2 : span要素のclass属性を指定する。 初期値はstring。 Return:
  • クォートで始まっていなければ1を返す。
sh
#------------------------------------------------------------------------------
function processSimpleQuote {
  local q=${1:-'&quot;'} class=${2:-string}
  if ! processStartSpanIf $q $class; then return 1; fi
  while feedBuffer; do
    if processEndSpanIf $q; then break; fi
    processSimpleEscapeSequence || processChar
  done
}

一単語処理する。

Params:

  • $1 : 単語を構成する一文字にマッチする正規表現。 初期値は[a-zA-Z0-9_]

Return:

  • 先頭文字が$1に含まれなければ1を返す。
sh
#------------------------------------------------------------------------------
function processWord {
  local r=${1:-'[a-zA-Z0-9_]'}
  processMatch "^$r+"
  return $?
}

キーワードを処理する。

Params:

  • $1 : キーワードを"|"で繋いだ文字列。
  • $2 : キーワードを括るspan要素のclass属性。初期値は"keyword"。

Return:

  • キーワードでなければ1を返す。
sh
#------------------------------------------------------------------------------
function processKeywords {
  local keys=$1 class=${2:-keyword}
  processMatch "^($keys)[[:>:]]" "<span class=\"$class\">" "</span>"
  return $?
}

シンプルな数値を処理する。

Params:

  • $1 : 数値だった場合のクラス名

Return:

  • 数値でなければ1を返す。
sh
#------------------------------------------------------------------------------
function processSimpleNumber {
  local class=${1:-number}
  if ! processMatch '^[-+.]?[0-9]' "<span class=\"$class\">"; then
    return 1;
  fi
  processMatch '^[-+.xXpP0-9a-fA-F]+'
  print -rn - '</span>'
}

出力行を処理する。

行頭は1行コメントなので先にこちらを処理しないとヒットしない。 Params:

  • $1 : 出力行の開始パターンを指定する。初期値は"#>"
  • $2 : 出力行が括られるspan要素のclass属性。初期値は"output"。

Return:

  • 出力行でなければ1を返す。
sh
#------------------------------------------------------------------------------
function processOutput {
  local header=${1:-'#&gt;'} class=${2:-output}
  if [[ $buf != $header* ]]; then return 1; fi
  print -rn - "<span class=\"$class\">"
  buf=${buf:${#header}}
  while [[ -n $buf ]]; do
    processSimpleNumber || \
    processChar
  done
  print -rn - '</span>'
}

function processVariableReferenceOfSh {
  local class=${1:-variable}
  # $" は変数ではない。
  if [[ $buf = \$\&quot\;* ]]; then return 1; fi
  if ! processStartSpanIf '$' $class; then return 1; fi
  processBalancedBracketOfSh || \
  processEntity || \
  processWord || \
  processChar

function processBalancedBracketOfSh {
  processBalancedOfSh || \

  return $?

function processShTokenInVariableReference {
  processVariableReferenceOfSh || \
  processSingleQuoteOfSh || \
  processDoubleQuoteOfSh || \
  processSimpleEscapeSequence || \
  processSimpleNumber || \
  processWord || \
  processChar

対応する括弧を処理する。

Params:

  • $1 : 開き括弧。初期値は'{'
  • $2 : 閉じ括弧。初期値は'}'

Return:

  • 開き括弧でない場合に1を返す。
sh
#------------------------------------------------------------------------------
function processBalancedOfSh {
  local open=${1:-\{} close=${2:-\}}
  if ! processIf $open; then return 1; fi
  while feedBuffer; do
    if processIf $close; then
      break;
    else
      processBalancedBracketOfSh || \
      processShTokenInVariableReference
    fi
  done
}

#------------------------------------------------------------------------------
function processBody {
  local language pre tag rest
  local -a code # コード行を貯める。
  while IFS= read -r rest; do
    while [[ -n $rest ]]; do
      # タグとそれ以外に分離する。
      if [[ ! $rest =~ "^([^<]*)(<[^>]+>)?(.*)" ]]; then continue; fi
      pre=$match[1]; tag=$match[2]; rest=$match[3]
      # code要素が来たら
      if [[ $tag =~ "^<code[[:space:]]+class[[:space:]]*=[[:space:]]*\"language-([^\"]*)\"[[:space:]]*>$" ]]; then
        language=${match[1]:-sh}
        print -rn - "<div class=\"language-type\">$language</div>";
        print -rn - $pre$tag
        code=($rest)
        while IFS= read -r rest; do
          if [[ ! $rest =~ "^([^<]*)(<[^>]+>)?(.*)" ]]; then continue; fi
          pre=$match[1]; tag=$match[2]; rest=$match[3]
          code+=$pre
          # /codeタグが来たらlanguageを終わる。
          if [[ $tag =~ "^</code[[:space:]]*>$" ]]; then
            print -lr - $code | processCode $language
            print -rn - $tag
            break;
          fi
        done
      else
        print -rn - $pre$tag
      fi
    done
    print -
  done
}
