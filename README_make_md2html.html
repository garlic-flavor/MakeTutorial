試験実装

概要

MarkdownからHTMLを生成する。

動機

  • MacOSでMarkdownを開くデフォルトアプリをXCodeから変えられない。
  • cmark-gfmの出力はコードのシンタックスハイライトがないので、 そういうものを付ける。
  • とりあえずHTML生成に対応しておけば何かいいことがあるかも知れない。

目標

  • 単一でJavascriptを使用しないHTMLを生成する。

謝辞

  • cmark-gfmを使っています。
sh

バッファが空なら埋める。

もしバッファが空なら、改行を出力し、バッファに読み込む。 Params:

  • -n: 改行を挿入しない。 Returns:
  • 読み込めるものがなければ1を返す。
sh
#------------------------------------------------------------------------------
function feedBuffer {
  local opt nonewline
  while getopts n opt; do
    case $opt in
      n) nonewline=true ;;
      *) :;;
    esac
  done
  (( 1 < OPTIND )) && shift $((OPTIND - 1))
  if [[ -z $buf ]] then
    if [[ ! $nonewline ]]; then
      print -
    fi
    if ! IFS= read -r buf; then
      return 1
    fi
  fi
}

#------------------------------------------------------------------------------
function processBody {
  local language rest pre tag
  local -a code
  while IFS= read -r rest; do
    while [[ -n $rest ]]; do
      # タグとそれ以外に分離する。
      if [[ ! $rest =~ "^([^<]*)(<[^>]+>)?(.*)" ]]; then continue; fi
      pre=$match[1]; tag=$match[2]; rest=$match[3]
      # code要素が来たら
      if [[ $tag =~ "^<code[[:space:]]+class[[:space:]]*=[[:space:]]*\"language-([^\"]*)\"[[:space:]]*>$" ]]; then
        language=${match[1]:-sh}
        print -rn - "<div class=\"language-type\">$language</div>";
        print -rn - $pre$tag
        code=($rest)
        while IFS= read -r rest; do
          if [[ ! $rest =~ "^([^<]*)(<[^>]+>)?(.*)" ]]; then continue; fi
          pre=$match[1]; tag=$match[2]; rest=$match[3]
          code+=$pre
          # /codeタグが来たらlanguageを終わる。
          if [[ $tag =~ "^</code[[:space:]]*>$" ]]; then
            print -lr - $code | processCode $language
            print -rn - $tag
            break;
          fi
        done
      else
        print -rn - $pre$tag
      fi
    done
    print -
  done
}
